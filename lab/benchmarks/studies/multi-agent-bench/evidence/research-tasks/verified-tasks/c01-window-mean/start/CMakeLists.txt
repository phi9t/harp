cmake_minimum_required(VERSION 3.16)
project(c01_window_mean LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(window src/window.cpp)
target_include_directories(window PUBLIC include)
add_executable(window_test tests/window_test.cpp)
target_link_libraries(window_test PRIVATE window)
enable_testing()
add_test(NAME window_test COMMAND window_test)
